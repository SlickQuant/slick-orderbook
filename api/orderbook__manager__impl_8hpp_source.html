47template<typename OrderBookT>
48template<typename Projection>
49auto OrderBookManager<OrderBookT>::findOrCreate(SymbolId symbol, Projection&& proj) {
50 // First, try read-only access (shared lock) - common case
51 {
52 std::shared_lock lock(mutex_);
53 auto it = symbol_map_.find(symbol);
54 if (it != symbol_map_.end()) [[likely]] {
55 return proj(it->second);
56 }
57 }
58
59 // Symbol doesn't exist - need exclusive lock to insert
60 std::unique_lock lock(mutex_);
61
62 // Double-check pattern: another thread might have inserted while we waited
63 auto it = symbol_map_.find(symbol);
64 if (it == symbol_map_.end()) {
65 it = symbol_map_.emplace(symbol, std::make_shared<OrderBookT>(symbol)).first;
66 }
67 return proj(it->second);
68}
